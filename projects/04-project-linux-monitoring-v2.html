Project 04: Linux Monitoring v2.0

Project Overview

Этот проект переводит навыки администрирования на уровень системного инжиниринга. Основные задачи: написание генераторов мусорных файлов и логов для проверки отказоустойчивости, анализ логов веб-сервера и развертывание полноценного мониторинга с дашбордами в Grafana.

Key Accomplishments

  • File & Log Generation: Написаны скрипты для автоматического заполнения файловой системы и генерации случайных логов Nginx.
  • System Cleanup: Реализованы три алгоритма очистки системы: по лог-файлу, по дате/времени и по маске имени.
  • Log Analysis & GoAccess: Настроена обработка логов Nginx для получения статистики через консольные скрипты и утилиту GoAccess.
  • Infrastructure Monitoring: Развернут стек Prometheus и Grafana с кастомными дашбордами для CPU, RAM и Disk.
  • Stress Testing: Проведены тесты под нагрузкой (stress-ng) для анализа поведения системы.

# file.sh (Генератор файлов)
# Функция создания файла с проверкой свободного места
function create_file () {
    local path=$1
    local size=$FILE_SIZE
    local name=$(basename $path | sed 's|\..*||')

    if (( $files_left > 0 )) ; then
        ((files_left--))
        local fullpath=$(readlink -f "${path}_$(date +%d%m%y)")
        echo "file ${fullpath} $(date +%y-%m-%d) ${FILE_SIZE} Kb" | tee -a "./01.log"
        
        # Проверка лимита (1 Гб) и создание файла
        df -h / | tail -n 1 | awk '{print $4}' | grep -q -e 'G$' && \
            dd if=/dev/zero of="${fullpath}" bs=1K count=${size} 2>/dev/null 1>/dev/null || \
            exit 1
    fi
}
# generate_name.sh (Генератор имен)
# Генерирует уникальное имя по списку букв
generate_name() {
    local chars="$1"   # Набор букв
    local minlen="$2"  # Минимальная длина
    local ext="$3"     # Расширение
    local name=""
    local used_chars=""
    
    while [ ${#used_chars} -lt ${#chars} ] || [ ${#name} -lt $minlen ]; do
        local current_char="${chars:${#used_chars}:1}"
        name+="$current_char"
        [[ $((RANDOM % 2)) -eq 0 ]] && name+="$current_char"
        used_chars+="$current_char"
        
        if [ ${#used_chars} -ge ${#chars} ] && [ ${#name} -lt $minlen ]; then
            name+="${chars:$((RANDOM % ${#chars})):1}"
        fi
    done

    local final_name="${name}_$(date +%d%m%y)"
    [[ -n "$ext" ]] && final_name+=".$ext"
    echo "$final_name"
}
# clean_by_log.sh (Очистка по журналу)
# Чтение лога и удаление объектов
while IFS=$'\t' read -r date type path size; do
    [[ "$type" == "file" ]] && files+=("$path")
    [[ "$type" == "dir" ]] && dirs+=("$path")
done < <(grep -v '^---' "$LOG_FILE")

# Удаление файлов
for f in "${files[@]}"; do
    [ -f "$f" ] && rm -f "$f" && ((REMOVED_FILES++))
done

# Удаление папок (от вложенных к родительским)
for ((i=${#dirs[@]}-1; i>=0; i--)); do
    d="${dirs[$i]}"
    [ -d "$d" ] && rmdir "$d" 2>/dev/null && ((REMOVED_DIRS++))
done
# generate_log.sh (Генератор логов)
# Формат Nginx combined: $remote_addr - - [$time_local] "$request"...
for ((i=0; i<$NUM_RECORDS; i++)); do
    ip=$(random_ip)
    method=$(random_method)
    code=$(random_code)
    url=$(random_url)
    agent=$(random_agent)
    time=$(random_time "$BASE_DATE" "$i")
    size=$((RANDOM % 5000 + 200))
    
    echo "$ip - - [$time] \"$method $url HTTP/1.1\" $code $size \"-\" \"$agent\"" >> "$LOG_FILE"
done
